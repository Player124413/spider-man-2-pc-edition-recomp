# Харнесс дифф-тестов для каждого сгенерированного модуля вместе с модулями, из которых он импортирует
# (локальный build/recon/imports.txt; внешние DLL — через HLE-заглушки).
set(deps_core msvcr70)
set(deps_engine core msvcr70)
set(deps_window engine core msvcr70)
set(deps_windrv window engine core msvcr70)
set(deps_game_exe window engine core msvcr70)
foreach(s d3ddrv fire gui ipdrv kwgame alaudio)
  set(deps_${s} engine core msvcr70)
endforeach()

foreach(s ${SNX_GEN_STEMS})
  if(NOT TARGET snx_gen_${s} OR NOT TARGET snx_gen_hle_stubs)
    continue()
  endif()
  set(mods "&mod_${s}")
  set(libs snx_gen_${s})
  set(complete TRUE)
  foreach(d ${deps_${s}})
    if(NOT TARGET snx_gen_${d})
      set(complete FALSE)
    endif()
    set(mods "&mod_${d}, ${mods}")
    list(APPEND libs snx_gen_${d})
  endforeach()
  if(NOT complete)
    continue()
  endif()
  set(hdr "${CMAKE_CURRENT_BINARY_DIR}/${s}/difftest_modules.h")
  string(REPLACE "&" "" decls "${mods}")
  string(REPLACE ", " ";" decl_list "${decls}")
  set(content "#pragma once\n#include \"runtime/loader.h\"\n")
  foreach(d ${decl_list})
    string(APPEND content "extern const rt_module_desc ${d};\n")
  endforeach()
  string(APPEND content "static const rt_module_desc *difftest_modules[] = {${mods}, 0};\n")
  file(WRITE "${hdr}.tmp" "${content}")
  configure_file("${hdr}.tmp" "${hdr}" COPYONLY)
  add_executable(difftest_${s} harness.c)
  target_include_directories(difftest_${s} PRIVATE ${PROJECT_SOURCE_DIR} "${CMAKE_CURRENT_BINARY_DIR}/${s}")
  target_link_libraries(difftest_${s} PRIVATE ${libs} snx_gen_hle_stubs snx_runtime)
endforeach()
