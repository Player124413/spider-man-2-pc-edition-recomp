# Загрузка игры без графики: образы всех модулей, DllMain по зависимостям, точка входа webhead.exe.
set(boot_mods "")
set(boot_libs "")
foreach(s ${SNX_GEN_STEMS})
  if(TARGET snx_gen_${s})
    list(APPEND boot_mods "&mod_${s}")
    list(APPEND boot_libs snx_gen_${s})
  endif()
endforeach()
set(snx_no_game OFF)
if(NOT boot_mods OR NOT TARGET snx_gen_hle_stubs)
  if(NOT ANDROID)
    return()
  endif()
  # Android без сгенерированного кода (CI без файлов игры): libmain.so из всего кода порта, но без модулей игры —
  # APK ставится, работают лаунчер и редактор управления, «Играть» сообщает, что игры в сборке нет
  message(WARNING "Нет модулей игры в ${SNX_GEN_DIR}: собирается APK без игры (SNX_NO_GAME)")
  set(snx_no_game ON)
  set(boot_mods "")
endif()
set(content "#pragma once\n#include \"runtime/loader.h\"\n")
foreach(m ${boot_mods})
  string(REPLACE "&" "" d "${m}")
  string(APPEND content "extern const rt_module_desc ${d};\n")
endforeach()
string(REPLACE ";" ", " mods_list "${boot_mods}")
if(snx_no_game)
  set(mods_list "0")
  set(boot_libs "")
else()
  string(APPEND mods_list ", 0")
endif()
string(APPEND content "static const rt_module_desc *const boot_modules[] = {${mods_list}};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/boot_modules.h.tmp" "${content}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/boot_modules.h.tmp" "${CMAKE_CURRENT_BINARY_DIR}/boot_modules.h" COPYONLY)
# иконка NRO — из локальной обложки или заставки игры (tools/make_icon.sh), один раз при конфигурации
set(boot_icon "${PROJECT_SOURCE_DIR}/build/icon/shrek2.jpg")
if(NINTENDO_SWITCH AND NOT EXISTS "${boot_icon}")
  execute_process(COMMAND sh "${PROJECT_SOURCE_DIR}/tools/make_icon.sh" "${boot_icon}" RESULT_VARIABLE icon_rc)
endif()
if(SNX_VARIANT STREQUAL "rus")
  set(boot_title "Шрек 2")
else()
  set(boot_title "Spider-Man 2")
endif()
set(boot_extra "")
if(ANDROID)
  list(APPEND boot_extra android.c)  # JNI: сенсорное управление и состояние игры для лаунчера (android/)
endif()
snx_add_app(snx_boot TITLE "${boot_title}" ICON "${boot_icon}" SOURCES main.c input.c prof.c config.c ${boot_extra}
            $<TARGET_OBJECTS:snx_hle>
            LIBS ${boot_libs} $<TARGET_NAME_IF_EXISTS:snx_gen_hle_stubs> snx_runtime snx_sdl2 snx_audio snx_font)
target_include_directories(snx_boot PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# горячие функции подряд в начале .text (порядок — tools/probe/hotorder.py по профилю SNX_PROF): кеш инструкций и
# TLB у A57 маленькие, а машинного кода ≈18 МБ с плоским профилем. По умолчанию — build/hot_order.ld, если есть.
set(SNX_SECTION_ORDER "${PROJECT_SOURCE_DIR}/build/hot_order.ld" CACHE FILEPATH "Порядок секций для ld (Switch)")
if(NINTENDO_SWITCH AND EXISTS "${SNX_SECTION_ORDER}")
  target_link_options(snx_boot PRIVATE "-Wl,--section-ordering-file=${SNX_SECTION_ORDER}")
  set_property(TARGET snx_boot APPEND PROPERTY LINK_DEPENDS "${SNX_SECTION_ORDER}")
endif()
if(SNX_VARIANT STREQUAL "rus")
  target_compile_definitions(snx_boot PRIVATE SNX_VARIANT_RUS)
endif()
if(snx_no_game)
  target_compile_definitions(snx_boot PRIVATE SNX_NO_GAME)
endif()
