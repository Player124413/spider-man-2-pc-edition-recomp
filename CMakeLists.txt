cmake_minimum_required(VERSION 3.20)
project(webheadNX VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(SNX_SANITIZE "" CACHE STRING "Санитайзеры компилятора (например, address;undefined, ON или OFF)")
if(SNX_SANITIZE AND NOT SNX_SANITIZE MATCHES "^(OFF|FALSE|0|NO)$")
  if(SNX_SANITIZE MATCHES "^(ON|TRUE|1|YES)$")
    set(_snx_san "address,undefined")
  else()
    string(REPLACE ";" "," _snx_san "${SNX_SANITIZE}")
    string(REPLACE " " "," _snx_san "${_snx_san}")
  endif()
  add_compile_options("-fsanitize=${_snx_san}" -fno-omit-frame-pointer)
  add_link_options("-fsanitize=${_snx_san}")
  if(_snx_san MATCHES "undefined")
    add_compile_options(-fno-sanitize-recover=undefined)
  endif()
  message(STATUS "webheadNX sanitizers: ${_snx_san}")
endif()

if(ANDROID)
  include(cmake/android.cmake)
elseif(NINTENDO_SWITCH)
  set(SNX_PLATFORM switch)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  add_library(snx_sdl2 INTERFACE)
  # libglad.pc содержит буквальный `$(DEVKITPRO)`, поэтому glad линкуем напрямую
  target_link_libraries(snx_sdl2 INTERFACE glad PkgConfig::SDL2)
  pkg_check_modules(SNXAUDIO REQUIRED IMPORTED_TARGET openal vorbisfile libavcodec libavutil)
  add_library(snx_audio INTERFACE)
  target_link_libraries(snx_audio INTERFACE PkgConfig::SNXAUDIO stdc++)  # openal-soft — C++
  pkg_check_modules(SNXFONT REQUIRED IMPORTED_TARGET freetype2)  # надписи слоя интерфейса (runtime/hle/overlay.c)
  add_library(snx_font INTERFACE)
  target_link_libraries(snx_font INTERFACE PkgConfig::SNXFONT)
elseif(APPLE)
  set(SNX_PLATFORM mac)
  find_package(PkgConfig QUIET)
  set(_snx_prefixes "")
  if(DEFINED ENV{HOMEBREW_PREFIX} AND EXISTS "$ENV{HOMEBREW_PREFIX}")
    list(APPEND _snx_prefixes "$ENV{HOMEBREW_PREFIX}")
  endif()
  list(APPEND _snx_prefixes /opt/homebrew /usr/local)
  list(REMOVE_DUPLICATES _snx_prefixes)
  set(_snx_inc_hints "")
  set(_snx_lib_hints "")
  foreach(_p ${_snx_prefixes})
    list(APPEND _snx_inc_hints "${_p}/include")
    list(APPEND _snx_lib_hints "${_p}/lib")
  endforeach()

  find_package(SDL2 CONFIG QUIET HINTS ${_snx_prefixes})
  if(SDL2_FOUND)
    set(_snx_sdl2_target SDL2::SDL2)
  elseif(PkgConfig_FOUND)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(_snx_sdl2_target PkgConfig::SDL2)
  else()
    find_package(SDL2 CONFIG REQUIRED HINTS ${_snx_prefixes})
    set(_snx_sdl2_target SDL2::SDL2)
  endif()
  find_package(OpenGL REQUIRED)
  add_library(snx_sdl2 INTERFACE)
  target_link_libraries(snx_sdl2 INTERFACE ${_snx_sdl2_target} OpenGL::GL)
  target_compile_definitions(snx_sdl2 INTERFACE GL_SILENCE_DEPRECATION)
  # звук и ролики: OpenAL системный, vorbisfile/vorbis/ogg и libavcodec/libavutil — из Homebrew, /usr/local или pkg-config
  find_library(OPENAL_FRAMEWORK OpenAL REQUIRED)
  if(PkgConfig_FOUND)
    pkg_check_modules(PC_VORBIS QUIET vorbisfile vorbis ogg)
    pkg_check_modules(PC_FFMPEG QUIET libavcodec libavutil)
    pkg_check_modules(PC_FREETYPE QUIET freetype2)
  endif()
  find_path(VORBIS_INCLUDE vorbis/vorbisfile.h HINTS ${PC_VORBIS_INCLUDE_DIRS} ${_snx_inc_hints} REQUIRED)
  find_library(VORBISFILE_LIB NAMES libvorbisfile.a vorbisfile HINTS ${PC_VORBIS_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  find_library(VORBIS_LIB NAMES libvorbis.a vorbis HINTS ${PC_VORBIS_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  find_library(OGG_LIB NAMES libogg.a ogg HINTS ${PC_VORBIS_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  find_path(AVCODEC_INCLUDE libavcodec/avcodec.h HINTS ${PC_FFMPEG_INCLUDE_DIRS} ${_snx_inc_hints} REQUIRED)
  find_library(AVCODEC_LIB NAMES avcodec HINTS ${PC_FFMPEG_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  find_library(AVUTIL_LIB NAMES avutil HINTS ${PC_FFMPEG_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  add_library(snx_audio INTERFACE)
  target_include_directories(snx_audio INTERFACE ${VORBIS_INCLUDE} ${AVCODEC_INCLUDE})
  target_link_libraries(snx_audio INTERFACE ${OPENAL_FRAMEWORK} ${VORBISFILE_LIB} ${VORBIS_LIB} ${OGG_LIB}
                        ${AVCODEC_LIB} ${AVUTIL_LIB})
  # надписи слоя интерфейса (runtime/hle/overlay.c): FreeType (ищем freetype/config/ftheader.h, чтобы не
  # подхватить неполный ft2build.h из сторонних macOS-фреймворков вроде Mono.framework)
  find_path(FREETYPE_INCLUDE freetype/config/ftheader.h HINTS ${PC_FREETYPE_INCLUDE_DIRS} ${_snx_inc_hints}
            PATH_SUFFIXES freetype2 REQUIRED)
  find_library(FREETYPE_LIB NAMES freetype HINTS ${PC_FREETYPE_LIBRARY_DIRS} ${_snx_lib_hints} REQUIRED)
  add_library(snx_font INTERFACE)
  target_include_directories(snx_font INTERFACE ${FREETYPE_INCLUDE})
  target_link_libraries(snx_font INTERFACE ${FREETYPE_LIB})
elseif(UNIX AND NOT APPLE)
  set(SNX_PLATFORM linux)
  find_package(PkgConfig REQUIRED)
  find_package(Threads REQUIRED)
  add_compile_options(-pthread)
  add_link_options(-pthread)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(GL REQUIRED IMPORTED_TARGET gl)
  add_library(snx_sdl2 INTERFACE)
  target_link_libraries(snx_sdl2 INTERFACE PkgConfig::SDL2 PkgConfig::GL Threads::Threads)
  target_compile_definitions(snx_sdl2 INTERFACE GL_GLEXT_PROTOTYPES)
  pkg_check_modules(SNXAUDIO REQUIRED IMPORTED_TARGET openal vorbisfile libavcodec libavutil)
  add_library(snx_audio INTERFACE)
  target_link_libraries(snx_audio INTERFACE PkgConfig::SNXAUDIO)
  pkg_check_modules(SNXFONT REQUIRED IMPORTED_TARGET freetype2)  # надписи слоя интерфейса (runtime/hle/overlay.c)
  add_library(snx_font INTERFACE)
  target_link_libraries(snx_font INTERFACE PkgConfig::SNXFONT)
else()
  message(FATAL_ERROR "Поддерживаются только macOS, Linux, Nintendo Switch и Android")
endif()
message(STATUS "webheadNX platform: ${SNX_PLATFORM}")

include(cmake/snx.cmake)
include(cmake/gen.cmake)

add_subdirectory(runtime)
if(SNX_PIN)
  # Единая копия rt_icall_slow и пул pin_wrap_* в pinned-ABI (x19–x28), чтобы не дублировать их в каждом .c build/gen/
  target_sources(snx_runtime PRIVATE "${PROJECT_SOURCE_DIR}/runtime/icall_pin.c")
  set_source_files_properties("${PROJECT_SOURCE_DIR}/runtime/icall_pin.c" PROPERTIES COMPILE_OPTIONS "-ffixed-x28")
endif()
if(NINTENDO_SWITCH)
  # Статическая линковка в один NRO: hidden убирает лишние GOT-загрузки в PIC/PIE для рантайма и HLE
  target_compile_options(snx_runtime PRIVATE -fvisibility=hidden)
  target_compile_options(snx_hle PRIVATE -fvisibility=hidden)
endif()
if(NOT ANDROID)
  add_subdirectory(host/hello)
endif()
add_subdirectory(host/boot)
include(CTest)
if(BUILD_TESTING AND NOT NINTENDO_SWITCH AND NOT ANDROID)
  add_subdirectory(tests/difftest)
  add_subdirectory(tests/bink)
  add_subdirectory(tests/rt)
endif()
